% Persian edition of the resume: same class options, geometry, spacing and
% section order as english/nedamani-resume-en.tex, with the RTL-specific parts
% marked below.
%
% No `%!TEX program' magic comment on purpose: LaTeX Workshop honours it by
% bypassing the configured recipes, which sent the output to persian/ instead
% of persian/build. Without it the `XeLaTeX' recipe from .vscode/settings.json
% runs, exactly as for English.

% Based on the Awesome-CV template: https://github.com/posquit0/Awesome-CV

\documentclass[11pt, a4paper]{awesome-cv}
\geometry{left=1.4cm, top=.8cm, right=1.4cm, bottom=1.0cm, footskip=.5cm}

\setbool{acvSectionColorHighlight}{true}
\renewcommand{\acvHeaderSocialSep}{\enspace\textbar\enspace}

%-------------------------------------------------------------------------------
% TYPOGRAPHY
%-------------------------------------------------------------------------------
% microtype must come before xepersian, which has to be the last package.
\usepackage{microtype}

% No mid-word breaks: a CV is a list of short items and proper nouns, so
% "architec- / tures" reads much worse than a slightly looser line. Turning
% hyphenation off alone would give TeX no legal break point and let lines run
% into the margin, so \emergencystretch lets the word spaces absorb the slack.
\hyphenpenalty=10000    % no automatic mid-word breaks
\exhyphenpenalty=10000  % never break at an existing hyphen (front-end, AI-assisted, ...)
\tolerance=9999         % accept the stretched line breaks from above
\emergencystretch=1em   % glue available to TeX before a line would overflow

%-------------------------------------------------------------------------------
% RTL, PERSIAN TYPOGRAPHY AND NUMERALS
%-------------------------------------------------------------------------------
% xepersian must be loaded after every other package: it loads bidi, which
% patches the directionality primitives of the whole document.
\usepackage{xepersian}

% Vazirmatn is the only typeface of the document, Persian and Latin alike, so a
% sentence mixing Persian with TypeScript or React keeps one type design.
%
% Vazirmatn ships no italic file, yet the CV styles ask for \itshape (address)
% and \slshape (dates, locations). Those switches reach the *Persian* font as
% soon as a Persian glyph appears in an italicised run, so the italics must be
% faked here as well as in awesome-cv/fonts.sty, otherwise LaTeX silently falls
% back to upright and warns.
\settextfont[
  Scale=1,
  BoldFont=Vazirmatn-Bold,
  ItalicFont=Vazirmatn-Regular,
  ItalicFeatures={FakeSlant=0.2},
  BoldItalicFont=Vazirmatn-Bold,
  BoldItalicFeatures={FakeSlant=0.2},
  Script=Arabic
]{Vazirmatn}

\setlatintextfont[
  Scale=1,
  BoldFont=Vazirmatn-Bold,
  ItalicFont=Vazirmatn-Regular,
  ItalicFeatures={FakeSlant=0.2},
  BoldItalicFont=Vazirmatn-Bold,
  BoldItalicFeatures={FakeSlant=0.2}
]{Vazirmatn}

% Vazirmatn carries Latin digits, so keeping the digit font in the same family
% makes phone number, page number and Solar Hijri years share one type design.
% Persian prose still uses Persian digits, which cv/*.tex rely on.
\setdigitfont[
  BoldFont=Vazirmatn-Bold,
  ItalicFont=Vazirmatn-Regular,
  ItalicFeatures={FakeSlant=0.2},
  BoldItalicFont=Vazirmatn-Bold,
  BoldItalicFeatures={FakeSlant=0.2}
]{Vazirmatn}
